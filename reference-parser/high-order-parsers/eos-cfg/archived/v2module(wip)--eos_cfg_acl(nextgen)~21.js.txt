/*********************************************************************************
 * Module:			eos_cfg_acl
 * Version:		 3.0
 * Parameters:	deviceName: string
 * Description: Parses a single arista configuration file acls.	Does not flatten into per-entry (yet)
 * Caveats:	?
 * Output:			List of acl objects
 *	 {
 *		 "aclName": name of the access-list,
 *		 "type":		'standard', 'extended',
 *		 "seqNo":	string representing sequence number,
 *		 "action":	'permit', 'deny', 'remark'
 *		 "value":	ip network if 'action' is 'permit' or 'deny', or comment if 'remark'
 *	 }
 */
const eos_cfg_acl = (deviceName) => {
	/** SECTION 1: module metadata ****/
	const pmodule = {
		name: 'eos_cfg_acl',	 // used for module identification when using logging in this module (i.e. logResult(`[${pmodule.name}]: blah blah blah`)
		ver: 1.0,									// future capability
		usage: '',								// future capability to provide help (may not be used)
		desc: ``,
		returnSchema: null,
	}


	/** SECTION 2: local result helpers to make code more readable (arguably).	****
	 *	These are candidates for possibly moving into an external module 
	 */
	const _0 = (arr) => arr[0]		// these get values out of arrays at the supplied index
	const _1 = (arr) => arr[1]
	const _2 = (arr) => arr[2]
	const _3 = (arr) => arr[3]
	const isTrue = (res) => true	// tags the existence of a result as 'true' - for flags (i.e. if xyz is parsed, xyz is true)


	//	intra-line parsers...
	const eosName = regex(/^[a-zA-Z0-9_-]+/)	 // generic alphanumeric (plus underscores and dashes) used for vrfNames, routeMap names, aclNames, etc
	const aclName = eosName.named('routeMapName')
	const aclSeqNo = digits.named('aclSeqNo')


	// line-level parsers...
	const stdAcl_header	= parseLine('ip access-list standard', aclName).map((result)=>result[1]).named('stdAclHeader')
	const stdAclEntry_permit	=	parseLine( aclSeqNo.tag('seqNo'), 'permit', possibly(strng('host')), ipNetworkOrAddress.tag('prefix') ).named('stdAclPermit')
	const stdAclEntry_deny		=	parseLine( aclSeqNo.tag('seqNo'), 'deny', possibly(strng('host')), ipNetworkOrAddress.tag('prefix') ).named('stdAclDeny')


	const extAcl_header	= parseLine('ip access-list', aclName, at_eol).map((result)=>result[1]).named('extAclHeader')

	// ip, tcp, ospf, etc 
	const extAcl_proto = word

	const extAcl_net = oneOf(
		strng('any'),
		seqOf(strng('host'), ipAddress),
		ipNetworkOrAddress,
	)
	
	const extAcl_ports_src = seqOf(
		strng('eq'),
		except(extAcl_net, many(extAcl_proto).map( (protos)=>protos.join('|') )),
	)

	const extAcl_ports_dst = seqOf(
		strng('eq'),
		many(extAcl_proto)	.map( (protos)=>protos.join('|')),
	)

	const restOf_extAcl	=	seqOf( 
		extAcl_proto.tag('proto'), 
		extAcl_net.tag('srcNet'),
		extAcl_ports_src.tag('srcPorts'),
		extAcl_net.tag('dstNet'),
		extAcl_ports_dst.tag('dstPorts'),
	)

	const statistic_perEntry = parseLine('statistics per-entry')
	const extAcl_deny		=	parseLine( aclSeqNo.tag('seqNo'), 'deny', restOf_extAcl.tag('denied') )
	//const extAcl_permit	=	parseLine( aclSeqNo.debugIf(({isError})=>isError).tag('seqNo'), 'permit', restOf_extAcl.tag('permitted') )
	const extAcl_permit	=	parseLine( aclSeqNo.tapOnError(`jager`).tag('seqNo'), 'permit', restOf_extAcl.tag('permitted') )

	const acl_remark			=	parseLine( aclSeqNo.tag('seqNo'), 'remark', rol.tag('remark') ).named('aclRemark')
	const unknown			 		= parseLine(rol).map(_0).named('unknown').tag('unknown')


	const stdAcl_entry = oneOf(
		stdAclEntry_permit,
		stdAclEntry_deny,
		acl_remark,
		unknown.logResult( (line) => (`${pmodule.name}, [stdAcl_entry]: the line '${line}' does not match any parser in the list`) ),
	)

	const extAcl_entry = oneOf(
		statistic_perEntry,
		extAcl_permit.named('extAcl_permit'), //.debugIf(({target,index,error}) => !error || !target.slice(index).startsWith('hi')),
		extAcl_deny.named('extAcl_deny'),
		acl_remark,
		unknown.logResult( (line) => (`${pmodule.name}, [extAcl_entry]: the line '${line}' does not match any parser in the list`) ),
	)


	// record-level parser 
	const stdAcl_record = seqOf(
		lookAhead(strng('ip access-list standard')),
		parseSection(seqOf(
			stdAcl_header.tag('aclName'),
			succeedWith('std').tag('aclType'),
			many(stdAcl_entry).tagList('aclEntries'),
		)),
	).getTaggedR()
	.map(multiplyBy('aclEntries'))
	.logEnv( ({
		section_index_start,
		section_index_parseduntil, 
		section_index_termination,
		section_text_unparsed,
		section_parser_dataLog,
	}) => ([
		// if the section parser has logs, make sure we pull them from the environment, but return empty if not
		(section_parser_dataLog?.length > 0) ? [
			`[module:${pmodule.name}, parser:stdAcl_record]: The section parser generated the following logs (note that indices below are relative to the parent index ${section_index_start}:`,
			section_parser_dataLog.map( (entry) => `  ${entry}` ),
		] : [],
		// if the section parser skips text for any reason, report it...
		(section_text_unparsed?.length > 0) ? [
			`[module:${pmodule.name}, parser:stdAcl_record]: The section parser did not parse entire section, starting from index ${section_index_termination}, and stopping at ${section_index_parseduntil}.  Skipped text is below:`,
			`  '${section_text_unparsed}'`,
		] : [],
	]))


	// record-level parser 
	const extAcl_record = seqOf(
		parseSection(seqOf(
			extAcl_header.tag('aclName'),
			succeedWith('ext').tag('aclType'),
			many(extAcl_entry).tagList('aclEntries'),
		)),
	).getTaggedR()
	.map(multiplyBy('aclEntries'))
	.logEnv( ({
		section_index_start,
		section_index_parseduntil, 
		section_index_termination,
		section_text_unparsed,
		section_parser_dataLog,
	}) => ([
		// if the section parser has logs, make sure we pull them from the environment, but return empty if not
		(section_parser_dataLog?.length > 0) ? [
			`[module:${pmodule.name}, parser:extAcl_record]: The section parser generated the following logs (note that indices below are relative to the parent index ${section_index_start}:`,
			section_parser_dataLog.map( (entry) => `  ${entry}` ),
		] : [],
		// if the section parser skips text for any reason, report it...
		(section_text_unparsed?.length > 0) ? [
			`[module:${pmodule.name}, parser:extAcl_record]: The section parser did not parse entire section, starting from index ${section_index_termination}, and stopping at ${section_index_parseduntil}.  Skipped text is below:`,
			`  '${section_text_unparsed}'`,
		] : [],
	]))


	const acl_record = oneOf(
		stdAcl_record,
		extAcl_record,
	)



	return seqOf(
		gotoSection('ip access-list'),
		many(acl_record).map(flattenArray)
	).map(res=>res[1])

}
